Private Sub DelPins()
'Urban 15.06.2022
On Error GoTo DelFehler
Pinstart = 0
PinEnd = 0

Pin2DelStart = Range("StartPin").Value
StkplatzNameStart = Left(Pin2DelStart, 1)
PinNameStart = Mid(Pin2DelStart, 2)
Pin2DelEnd = Range("EndPin").Value
StkplatzNameEnd = Left(Pin2DelEnd, 1)
PinNameEnd = Mid(Pin2DelEnd, 2)
' Endpin mit Wert belegen
If Pin2DelStart <> "m" And Pin2DelEnd = "n" Then
      Pin2DelEnd = Pin2DelStart
      StkplatzNameEnd = Left(Pin2DelEnd, 1)
      PinNameEnd = Mid(Pin2DelEnd, 2)
End If
' Startpin mit Wert belegen
If Pin2DelStart = "m" And Pin2DelEnd <> "n" Then
      Pin2DelStart = Pin2DelEnd
      StkplatzNameStart = Left(Pin2DelStart, 1)
      PinNameStart = Mid(Pin2DelStart, 2)
End If

'Datenumwandlung
If IsNumeric(PinNameStart) Then
      Pinstart = CInt(PinNameStart) ' Pinname als integer
End If
If IsNumeric(PinNameEnd) Then
      PinEnd = CInt(PinNameEnd) ' Pinname als integer
End If
' Start und Endpin in die richtige Reihenfolge bringen
If Pinstart > PinEnd And Not PinNameEnd = "" Then
   Pin2DelStart = Range("EndPin").Value
   Pin2DelEnd = Range("StartPin").Value
   PinNameStart = Mid(Pin2DelStart, 2)
   PinNameEnd = Mid(Pin2DelEnd, 2)
   'Datenumwandlung
      If IsNumeric(PinNameStart) Then
         Pinstart = CInt(PinNameStart) ' Pinname als integer
      End If
      If IsNumeric(PinNameEnd) Then
         PinEnd = CInt(PinNameEnd) ' Pinname als integer
      End If
   
End If
If Pin2DelStart = "m" And Pin2DelEnd = "n" Then
    info = MsgBox("Wrong Pin Definition, Exit!", vbInformation, "Fehler Pinbereich - Error Pin range")
    Exit Sub
End If

Steckplz2Del = StkplatzNameStart
'Datenumwandlung
      If IsNumeric(PinNameStart) Then
         Pinstart = CInt(PinNameStart) ' Pinname als integer
      End If
      If IsNumeric(PinNameEnd) Then
         PinEnd = CInt(PinNameEnd) ' Pinname als integer
      End If
           
 
If StkplatzNameStart <> StkplatzNameEnd Then
   info = MsgBox("Range of Pins out of one slot, Exit!", vbInformation, "Fehler Pinbereich - Error Pin range")
   Exit Sub
End If
If Pin2DelStart = "m" Then
   info = MsgBox("No Pin defined > nothing to do delete, Exit!", vbInformation, "Pinauswahl vergessen? - No Pin selection")
   Exit Sub
End If
If Pin2DelStart <> "m" And Pin2DelEnd <> "n" Then
   frage = Application.InputBox(Prompt:="Do you want to delete " & vbCrLf & " this Pin(s) [ " & Pin2DelStart & " - " & Pin2DelEnd & " ] (yes/no)?", Default:="yes", Type:=2)
   If frage = False Then Exit Sub
     If frage <> "yes" Then
        info = MsgBox("No Pin to do delete, Exit!", vbInformation)
        Range("GER").Value = "m"
        Exit Sub
     End If
   If frage = "yes" Then
    ' ermittelt Pinanzahl
        'PinsAnZahl = (Range("PinsLower").Row - 1) - (Range("PinsUpper").Row + 1)
        StrtPinNUm = Range("PinsUpper").Row + 1
        StartStplzCell = -1
        'Range("SumPins").Value = PinsAnZahl - PinCounter + 1
        PinsAnZahl = Range("SumPins").Value
        'erste Pinzeile finden
        For PiS = 0 To PinsAnZahl
           'PinBereich kann sich geändert haben, wird aktualisiert
            SteckplatzName = Cells(StrtPinNUm + PiS, 2).Value
            SteckplatzPin = Cells(StrtPinNUm + PiS, 7).Value
            'Umwandeln SteckplatzPin Urban 24.06.22
            If IsNumeric(SteckplatzPin) Then
               SteckplatzPin = CInt(SteckplatzPin) ' Pinname als integer
            End If
            If SteckplatzName = Steckplz2Del And SteckplatzPin = Pinstart Then
              Pin2DelStartCellRow = Cells(StrtPinNUm + PiS, 2).Row
            End If
            If SteckplatzName = Steckplz2Del And SteckplatzPin = PinEnd Then
              Pin2DelEndCellRow = Cells(StrtPinNUm + PiS, 2).Row
            End If
           Next

    'löscht PinZeilen
    If Pin2DelStartCellRow = "" Then Pin2DelStartCellRow = Pin2DelEndCellRow
    If Pin2DelEndCellRow = "" Then Pin2DelEndCellRow = Pin2DelStartCellRow
   'Pin2DelEndCellRow = Range(Pin2DelEnd).Row
    zwtxt = Pin2DelStartCellRow & ":" & Pin2DelEndCellRow
    PinCounter = Pin2DelEndCellRow - Pin2DelStartCellRow + 1
    If PinsAnZahl = PinCounter Then
      MsgBox ("Delete Error: Minimum PinCount is 1 Pin, Exit")
      Exit Sub
    End If
    If PinsAnZahl > 1 Then
       Rows(zwtxt).EntireRow.Delete
       Range("SumPins").Value = PinsAnZahl - PinCounter
       MsgBox ("Urgent: Please Check PinNames and Connectors! Use Funktion Pins_Generieren")
       Else: MsgBox ("One Pin is necessary can not delete!")
    End If
   End If
End If

'Bereich Pins und Slots wiederherstellen, da evtl. Steckplz A gelöscht wurde
ActiveWorkbook.Names.Add Name:="Pins", RefersToR1C1:="=Formblatt!R17C2:R17C230"
Bereich1Ger = "=Formblatt!$B$" & StrtGerNum & ":$CS$" & StrtGerNum
'ActiveWorkbook.Names.Add Name:="Slots", RefersTo:=Bereich1Ger
Range("StartPin").Value = "m"
Range("EndPin").Value = "n"
'Urban CheckMe 21.03.2023
  Tabelle1.CommandButton2.Caption = "CheckMe"
  Range("DataChecked").Value = "ungeprueft"
  Range("State").Value = ""
DelFehler:
Call ExChangeGER
Call ExChange
End Sub


Private Sub DelSteckplz_Click()
'Urban 01.02.2022
Steckplz2Del = Range("GER").Value
If Steckplz2Del = "n" Then
   info = MsgBox("Nothing to do delete, Exit!", vbInformation, "Stecker vergessen? - No plug selection")
   Range("GER").Value = "n"
   Exit Sub
End If
If Steckplz2Del <> "n" Then
   frage = Application.InputBox(Prompt:="Do you want to delete " & vbCrLf & " this Connector (Slot) [ " & Steckplz2Del & " ] (yes/no)?", Default:="yes", Left:=20, Top:=40, Type:=2)
     If frage = False Then Exit Sub
     If frage <> "yes" Then
        info = MsgBox("Nothing to do delete, Exit!", vbInformation)
        Range("GER").Value = "n"
        Exit Sub
     End If
     If frage = "yes" Then
        ' ermittelt Pinanzahl
        PinsAnZahl = (Range("PinsLower").Row - 1) - (Range("PinsUpper").Row + 1)
        StrtPinNUm = Range("PinsUpper").Row + 1
        StartStplzCell = -1
        'Abbruch bei Verletzung der Minimalbedingungen
        If (Range("SlotsUpper").Row + 1) - (Range("SlotsLower").Row - 1) = 0 Then
           info = MsgBox("Mindestens ein Steckplatz (A) muss vorhanden sein! Abbruch" & vbCrLf & "One Connector part is necessary minimum! Abort", vbInformation, "Minimalanforderungen werden verletzt")
           Range("GER").Value = "n"
           Exit Sub
        End If
        
           For PiS = 0 To PinsAnZahl
           'PinBereich kann sich geändert haben, wird aktualisiert
            SteckplatzName = Cells(StrtPinNUm + PiS, 2).Value
            SteckplatzPin = Cells(StrtPinNUm + PiS, 7).Value
            If SteckplatzName = Steckplz2Del Then
              PinCounter = PinCounter + 1
              tmpStplzCell = Cells(StrtPinNUm + PiS, 7).Row
                If StartStplzCell = -1 Then StartStplzCell = tmpStplzCell
                  If tmpStplzCell > StartStplzCell Then
                    EndStplzCell = Cells(StrtPinNUm + PiS, 7).Row
                  End If
            End If
           Next
        'Anzahl Steckplätzte und GesamtPinanzahl reduzieren
        ' PinCounter ist am Ende um einen Pin zu hochgezählt
          Range("SumPins").Value = PinsAnZahl - PinCounter + 1
           PinsAnZahl = Range("SumPins").Value
          Range("SumSlots").Value = Range("SumSlots").Value - 1
           SlotAnZahl = Range("SumSlots").Value
        'löscht PinZeilen
          If EndStplzCell > StartStplzCell Then
            zwtxt = StartStplzCell & ":" & EndStplzCell
            Else: zwtxt = StartStplzCell
          End If
          Rows(zwtxt).EntireRow.Delete
          Range("GER").Value = "n" ' Steckplz- Auswahl zurücksetzen
          ExChange  ' Pinzanzahl für Exchange aktualisieren
         'SteckplzTabelle korrigieren
           StrtGerNum = Range("SlotsUpper").Row + 1
           EndGerNum = Range("SlotsLower").Row - 1
           For i = StrtGerNum To EndGerNum
             tmpSteckplzName = Cells(i, 2).Value
             tmpGERrow = Cells(i, 2).Row
            'Steckerzeile löschen
             If tmpSteckplzName = Steckplz2Del Then
               Rows(tmpGERrow).Delete
             End If
           Next
            ExChangeGER ' SteckPlatzanzahl für DeleteSteckPlz aktualisieren
            'SteckPlatz Namen neu vergeben und Namentausch merken
            StrtGerNum = Range("SlotsUpper").Row + 1
            EndGerNum = Range("SlotsLower").Row - 1
            EndPinNUm = Range("PinsLower").Row ' Neuer EndPin durch Löschen
            cnt = 0
            For j = StrtGerNum To EndGerNum
              cnt = cnt + 1
              akt_SpSteckerN = Cells(j, 2).Value
              gen_SpSteckerN = Chr$(64 + cnt)
              If gen_SpSteckerN <> akt_SpSteckerN Then
                Cells(j, 2).Value = gen_SpSteckerN
              End If
            Next 'Steckerzeile umbenennen
            'am Pin SteckPlatz Namen neu vergeben dazu Speicher SpSteckerN nutzen
            GerCnt = 1
            int_Z = 1
            PinsAnZahl = (Range("PinsLower").Row - 1) - (Range("PinsUpper").Row + 1)
            For PiS = 0 To PinsAnZahl
               'genSteckerName = Chr$(64 + GerCnt)
               origSteckNanme = Cells(StrtPinNUm + PiS, 2).Value
               origSteckNPin = Cells(StrtPinNUm + PiS, 7).Value
                 Cells(StrtPinNUm + PiS, 2).Value = Chr$(64 + GerCnt)
                 'Schleife aller Pins des Steckers
                 For k = 1 To 300
                   innerSteckNanme = Cells(StrtPinNUm + PiS + k, 2).Value
                   innerSteckNPin = Cells(StrtPinNUm + PiS + k, 7).Value
                   If EndPinNUm = (StrtPinNUm + PiS + k) Then
                     ' Abbruch da ausserhalb der Pintabelle
                     Exit For
                   End If
                   If innerSteckNanme = "" Then innerSteckNanme = origSteckNanme
                   If origSteckNanme = innerSteckNanme Then
                      Cells(StrtPinNUm + PiS + k, 2).Value = Chr$(64 + GerCnt)
                   End If
                    If origSteckNanme <> innerSteckNanme Then
                      PiS = PiS + k - 1
                      Exit For
                    End If
                 Next
                 GerCnt = GerCnt + 1
               If EndPinNUm = (StrtPinNUm + PiS + k) Then
                    ' Abbruch da ausserhalb der Pintabelle
                     Exit For
               End If

     End If
End If

      If ReadyDB = "imported" And Aktuell <> "initial" Or ReadyDB = "imported" And Aktuell <> "#N/A" Then
          'UGU 23.06.2018, BGERtabelle füllen wenn die Datenbasis BGER Daten enthält (Altdaten haben keine)
          'sonst ein Fehlermeldung ausgeben und Import DB starten
          BGER_PinTab_UserFill (Aktuell) ' UGU 23.06.2018
          Vortrag = Aktuell
          Else:
             Application.StatusBar = "Die Datenbasis fehlt oder wurde nicht importiert, database is missing, or isn't imported"
             StarteImport_CSVdatei ' die BGER Tabelle kann erst gefüllt werden, wenn die Datenbasis importiert wurde
       End If
    End If
    Dim o_ActualCell As Range
    Dim o_MergedCell As Range
    Dim o_BackupCell As Range

  EndPinS = "AH" & Range("PinsLower").Row - 1    'lezter Pin

'#################################################################################

    MehrfachNio = False
'End If
'Ende: DropDown-Liste mit Mehrfachauswahl

    ' Wird schon geändert oder werden Pins generiert?
    If (b_InChange) Or (b_IsGeneratingPinsSlots) Then
        Exit Sub
    End If
    
    ' Start: Änderung
    b_InChange = True
    
    ' Umgang mit verbundenen Zellen und Mehrfachselektionen
    For Each o_ActualCell In o_Target
        ' Bestimme die Zelle links oben in verbundenen Zellen. Nur diese Zellen (oder nicht verbundene Zellen)
        ' werden im weiteren betrachtet.
        Set o_MergedCell = o_ActualCell.MergeArea
        If (o_ActualCell.Row = o_MergedCell.Row) And (o_ActualCell.Column = o_MergedCell.Column) Then
            ' Wird der korrekte Bereich editiert?
            If (Is_Pins_Row(o_ActualCell)) Then
                If (o_ActualCell.Column = 52) Then
                    ' Pintyp
                    Set o_BackupCell = Cells(o_ActualCell.Row, 140)
                    If (o_ActualCell.Value <> o_BackupCell.Value) Then
                        o_BackupCell.Value = o_ActualCell.Value
                        Call Edit_Pintype(o_Target, o_ActualCell)
                    End If
                ElseIf (o_ActualCell.Column = 65) Then
                    ' Pincharakteristik
                    Set o_BackupCell = Cells(o_ActualCell.Row, 141)
                    If (o_ActualCell.Value <> o_BackupCell.Value) Then
                        o_BackupCell.Value = o_ActualCell.Value
                        Call Edit_Pinchar(o_Target, o_ActualCell)
                    End If
                End If
            End If
        End If
    Next o_ActualCell
    
    ' Ende: Änderung
    b_InChange = False
End Sub
' Die übergebene Zelle entspricht einem Pintyp und wird bearbeitet
Private Sub Edit_Pintype(ByVal o_Target As Range, ByVal o_ActualCell As Range)
    ' Inhalt ändern: Pincharakteristik
    Dim o_CharCell As Range
    Set o_CharCell = Cells(o_ActualCell.Row, 65)
    If (Application.Intersect(o_CharCell, o_Target) Is Nothing) Then
        o_CharCell.Value = ""
    End If
    ' Stromzellen setzen
    Call Set_Current_Cells(o_ActualCell, False)
    ' Ggf. die Beschreibung (KL-Info) setzen
    If o_ActualCell.Value = "Nicht verbunden" Then
        Dim o_DescCell As Range
        Set o_DescCell = Cells(o_ActualCell.Row, 15)
        o_DescCell.Value = "n.c."
    End If
End Sub
' Die übergebene Zelle entspricht einer Pincharakteristik und wird bearbeitet
Private Sub Edit_Pinchar(ByVal o_Target As Range, ByVal o_ActualCell As Range)
    ' Wenn gleichzeitig der Pintyp geändert wird, dann müssen wir nichts tun
    Dim o_TypeCell As Range
    Set o_TypeCell = Cells(o_ActualCell.Row, 52)
    If (Application.Intersect(o_TypeCell, o_Target) Is Nothing) Then
        ' Stromzellen setzen
        Call Set_Current_Cells(o_TypeCell, False)
    End If
End Sub

Public Sub AddPins2Con()
On Error GoTo PinFehler
'Urban 01.02.2023
Pinstart = 0
PinEnd = 0
Dim o_RangeTmp1 As Range
Dim o_RangeTmp2 As Range
PinsGStat = True
Tabelle1.NEWpins = "edit"
Pin2InsertStart = Range("StartPin").Value
StkplatzNameStart = Left(Pin2InsertStart, 1)
PinNameStart = Mid(Pin2InsertStart, 2)
Pin2InsertEnd = Range("EndPin").Value
StkplatzNameEnd = Left(Pin2InsertEnd, 1)
PinNameEnd = Mid(Pin2InsertEnd, 2)
If Pin2InsertStart = "m" And Pin2InsertEnd = "n" Or Pin2InsertStart <> "m" And Pin2InsertEnd <> "n" Then
    info = MsgBox("Wrong Pin Definition, Exit!", vbInformation, "Nur einen Pin in [m] oder [n]auswählen - Error Pin Range")
    Range("StartPin").Value = "m"
    Range("EndPin").Value = "n"
    Exit Sub
End If
' Endpin mit Wert belegen
If Pin2InsertStart <> "m" And Pin2InsertEnd = "n" Then
      Pin2InsertEnd = Pin2InsertStart
      StkplatzNameEnd = Left(Pin2InsertEnd, 1)
      PinNameEnd = Mid(Pin2InsertEnd, 2)
End If
' Startpin mit Wert belegen
If Pin2InsertStart = "m" And Pin2InsertEnd <> "n" Then
      Pin2InsertStart = Pin2InsertEnd
      StkplatzNameStart = Left(Pin2InsertStart, 1)
      PinNameStart = Mid(Pin2InsertStart, 2)
End If

'Datenumwandlung
If IsNumeric(PinNameStart) Then
      Pinstart = CInt(PinNameStart) ' Pinname als integer
      Else: Pinstart = PinNameStart
End If

If IsNumeric(PinNameEnd) Then
      PinEnd = CInt(PinNameEnd) ' Pinname als integer
      Else: PinEnd = PinNameEnd
End If

' Start und Endpin in die richtige Reihenfolge bringen
If Pinstart > PinEnd And Not PinNameEnd = "" Then
   Pin2InsertStart = Range("EndPin").Value
   Pin2InsertEnd = Range("StartPin").Value
   PinNameStart = Mid(Pin2InsertStart, 2)
   PinNameEnd = Mid(Pin2InsertEnd, 2)
End If

Steckplz2Ins = StkplatzNameStart

If StkplatzNameStart <> StkplatzNameEnd Then
   info = MsgBox("Range of Pins out of one slot, Exit!", vbInformation, "Fehler Pinbereich - Error Pin range")
   Exit Sub
End If
If Pin2InsertStart = "m" Then
   info = MsgBox("No Pin defined > nothing to do, Exit!", vbInformation, "Pinauswahl vergessen? - No Pin selection")
   Exit Sub
End If
If Pin2InsertStart <> "m" And Pin2InsertEnd <> "n" Then
   frage = Application.InputBox(Prompt:="Insert neu Pin " & vbCrLf & " above Pin [ " & Pin2InsertStart & " ] (yes/no)?", Default:="yes", Type:=2)
   If frage = False Then Exit Sub
     If frage <> "yes" Then
        info = MsgBox("No Pin to do insert, Exit!", vbInformation)
        Range("GER").Value = "m"
        Exit Sub
     End If
   If frage = "yes" Then
    ' ermittelt Pinanzahl
        'PinsAnZahl = (Range("PinsLower").Row - 1) - (Range("PinsUpper").Row + 1)
        StrtPinNUm = Range("PinsUpper").Row + 1
        StartStplzCell = -1
        'Range("SumPins").Value = PinsAnZahl - PinCounter + 1
        PinsAnZahl = Range("SumPins").Value
        'erste Pinzeile finden
        For PiS = 0 To PinsAnZahl
           'PinBereich kann sich geändert haben, wird aktualisiert
            SteckplatzName = Cells(StrtPinNUm + PiS, 2).Value
            SteckplatzPin = Cells(StrtPinNUm + PiS, 7).Value
            'Umwandeln SteckplatzPin Urban 24.06.22
            If IsNumeric(SteckplatzPin) Then
               SteckplatzPin = CInt(SteckplatzPin) ' Pinname als integer
            End If
            If SteckplatzName = Steckplz2Ins And SteckplatzPin = Pinstart Then
              Pin2InsertStartCellRow = Cells(StrtPinNUm + PiS, 2).Row
            End If
            If SteckplatzName = Steckplz2Ins And SteckplatzPin = PinEnd Then
              Pin2InsertEndCellRow = Cells(StrtPinNUm + PiS, 2).Row
            End If
           Next

    ' PinZeile einfügen
    If Pin2InsertStartCellRow = "" Then Pin2InsertStartCellRow = Pin2InsertEndCellRow
    If Pin2InsertEndCellRow = "" Then Pin2InsertEndCellRow = Pin2InsertStartCellRow
   'Pin2InsertEndCellRow = Range(Pin2InsertEnd).Row
    zwtxt = Pin2InsertStartCellRow '& ":" & Pin2InsertEndCellRow
    PinCounter = Pin2InsertEndCellRow - Pin2InsertStartCellRow + 1
    If PinsAnZahl = PinCounter Then
      MsgBox ("Insert Error: Minimum PinCount is 1 Pin, Exit")
      Exit Sub
    End If
    
    If PinsAnZahl > 1 Then
       Rows(zwtxt).EntireRow.Insert Shift:=xlShiftDown
       Cells(Pin2InsertStartCellRow + PinCounter - 1, 1).Select
       Set o_RangeTmp1 = Range("Pins")
        i_StartRow = o_RangeTmp1.Row
        i_StartColumn = o_RangeTmp1.Column
        i_EndColumn = i_StartColumn + o_RangeTmp1.Columns.Count - 1
       Set o_RangeTmp2 = Range(Cells(Pin2InsertStartCellRow + PinCounter - 1, i_StartColumn), Cells(Pin2InsertStartCellRow + PinCounter - 1, 226))
        ' Eigentlicher Füllvorgang
        o_RangeTmp1.Copy
        o_RangeTmp2.PasteSpecial
        ' Ggf. vorhanden Kopierrahmen entfernen
        Application.CutCopyMode = False
        Cells(Pin2InsertStartCellRow + PinCounter - 1, 2).Select
        Cells(Pin2InsertStartCellRow + PinCounter - 1, 7).Value = "99"
        Cells(Pin2InsertStartCellRow + PinCounter - 1, 2).Value = StkplatzNameStart
        Range("SumPins").Value = PinsAnZahl + PinCounter
       MsgBox ("Urgent: Please Check PinNames and Connectors! Use Funktion Pins_Generieren")
       Else: MsgBox ("One Pin is necessary can not delete!")
    End If
   End If
End If
NEWpins = "fix"
PinsGStat = False ' sorgt dafür das Userform1 startet
Range("StartPin").Value = "m"
Range("EndPin").Value = "n"
Fehler:
Call ExChangeGER
Call ExChange
PinFehler:
End Sub
