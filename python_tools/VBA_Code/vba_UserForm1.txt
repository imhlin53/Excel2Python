Private Sub UserForm_Initialize()

LAH_TxB.Text = BN_LAH_ID
'Wortzerlegung
Len_AnBAnfg = Len(AnBAnfg) 'Länge des Kommentar feststellen
'Schleife findet Trennzeichen
AnB_pos = InStr(1, AnBAnfg, ";") 'Suche nach einem Trennzeichen ";" im String der Anbindungsanforderungen
If AnB_pos = 0 And Len_AnBAnfg >= 1 Then Text_Auswertung (AnBAnfg)
If AnB_pos = 0 And Len_AnBAnfg = 0 Then
   clearCB_Click
   Exit Sub
End If
i = 1
len_AnBtxt = 0
While AnB_pos > 0
'Text herausholen
AnB_pos = InStr(i, AnBAnfg, ";")
If AnB_pos > 0 Then AnB_pos_pre = AnB_pos

If AnB_pos <> 0 Then
  len_AnBtxt = AnB_pos - i
  tmp_Txt = Mid(AnBAnfg, i, len_AnBtxt)
End If

If AnB_pos = 0 And AnB_pos_pre > 0 Then
   'Ausnahme bei Länge Null
   len_AnBtxt = Len_AnBAnfg - AnB_pos_pre
   tmp_Txt = Mid(AnBAnfg, AnB_pos_pre + 1, len_AnBtxt)
End If
 i = AnB_pos + 1
 'tmp_Txt auswerten LayL, Qmin LayL40mm Qmin-0.82 ASIL_CB.Value
 Text_Auswertung (tmp_Txt) ' Sprung zur Funktion Text_Auswertung
 
Wend
 
End Sub

Public Function Text_Auswertung(tmp_Txt)

 TxtCut = Left(tmp_Txt, 4)
 testAnBind = InStr(1, TxtCut, "(")
 If testAnBind > 1 Then
   ASIL_CB.Value = tmp_Txt  'TxtCut = Left(TxtCut, testAnBind - 1) ' Neu urban 23.08.2023
 End If
 TxtCut1 = Left(tmp_Txt, 3)
 TxtCut2 = Right(tmp_Txt, 5)
 'MindestLtgQuerschnitt
 If TxtCut = "Qmin" Then QminCB.Value = tmp_Txt
 'Schirmung
 If TxtCut = "KOAX" Or TxtCut = "RG58" Or TxtCut = "RG17" Or TxtCut = "RTK0" Or TxtCut = "FLHV" Then SchirmungCB.Value = tmp_Txt
 If TxtCut1 = "UTP" Or TxtCut1 = "STP" Then SchirmungCB.Value = tmp_Txt
 'ASIL
 If TxtCut = "ASIL" Or TxtCut = "ADAP" Or TxtCut1 = "HAF" Or tmp_Txt = "QM" Then ASIL_CB.Value = tmp_Txt
 'Schlaglänge
 If TxtCut = "LayL" Then LayLengthCB.Value = tmp_Txt
 'Terminierung
 If TxtCut2 = "_Term" Then TerminierungCB.Value = tmp_Txt
 
End Function
